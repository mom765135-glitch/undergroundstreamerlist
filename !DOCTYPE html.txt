  return r.sort((a,b) => S.placement[a]-S.placement[b] || orderIdx(a)-orderIdx(b));
}

function removeStreamer(n){ S.hidden.push(k(n)); delete S.placement[n]; S.order=S.order.filter(x=>x!==n); save(); render(); }

let adding=false;
async function addStreamer(){
  if (adding) return;
  const inp=$("addIn"), avgIn=$("avgIn");
  const n = inp.value.trim().replace(/^https?:\/\/(www\.)?twitch\.tv\//i,"").replace(/[\/?#].*$/,"").replace(/^@/,"");
  if (!/^[A-Za-z0-9_]{2,25}$/.test(n)) { status("Enter a Twitch username like cloyouss."); return; }
  if (roster().some(x=>k(x)===k(n))) { status(n+" is already on the list."); return; }
  adding=true;
  try {
    status("Looking up "+n+"…");
    const pf = await resolvePfp(n);
    if (pf==="missing") { status("Couldn't find a Twitch account named "+n+"."); return; }
    let avg = await lookupAvg(n);
    if (avg===null && avgIn.value!=="") avg = Math.round(Number(avgIn.value));
    if (avg===null || isNaN(avg)) { status("Couldn't read "+n+"'s average viewers automatically. Type it in the Avg viewers box and press Add again."); avgIn.focus(); return; }
    if (avg>MAX_AVG) { status(n+" averages "+avg+" viewers, over the "+MAX_AVG+" cap, so they weren't added."); return; }
    S.hidden = S.hidden.filter(x=>x!==k(n)); S.added.push(n); S.avg[n]=avg; inp.value=""; avgIn.value=""; save();
    if (hasBackend()) { try { await sbUpsert("streamers",{name:n,avg},"name"); REMOTE.streamers.push({name:n,avg}); } catch(e){} }
    render(); status("Added "+n+" ("+avg+" avg viewers)."+(pf==="ok" ? "" : " Their PFP couldn't be loaded."));
  } finally { adding=false; }
}

function makeBallot(){
  const tiers = {}; roster().forEach(n => { if (S.placement[n]!==undefined) tiers[n]=S.placement[n]; });
  return { voter:S.vid, name:(S.name||"").trim()||"Anonymous", tiers, rank:ranking().slice(0,MIN_RANK) };
}
async function finish(){
  const b = makeBallot(); S.submitted=true; S.ballots[S.vid]=b; save();
  if (hasBackend()) { try { await sbUpsert("ballots",b,"voter"); await loadRemote(); } catch(e){ status("Your ballot is saved here, but the shared server didn't accept it."); } }
  showResults();
}
function show(id){ ["intro","app","results"].forEach(s => $(s).hidden = s!==id); window.scrollTo(0,0); }

function pickText(name, rank, myT, A){
  const parts = []; const l = A[name];
  const place = rank===1 ? "Your number one." : rank<=3 ? "On your podium at #"+rank+"." : rank===MIN_RANK ? "The last spot in your top "+MIN_RANK+"." : "#"+rank+" on your list.";
  parts.push(place+" You put them in "+L(myT)+" tier.");
  if (l && l.length>=2) {
    const st = stats(l), ct = Math.min(5,Math.max(0,Math.round(st.mean))), d = myT-st.mean;
    if (d < -0.75) parts.push("That's higher than the crowd, who land around "+L(ct)+" ("+l.length+" ballots).");
    else if (d > 0.75) parts.push("That's tougher than the crowd, who lean "+L(ct)+" ("+l.length+" ballots).");
    else parts.push("You and the community are close: "+L(ct)+" tier across "+l.length+" ballots.");
  } else parts.push("Nobody else has rated them yet, so you're setting the early mark.");
  if (S.avg[name]!==undefined) parts.push("They average about "+S.avg[name]+" viewers.");
  return parts.join(" ");
}
function communityText(e, mineRank){
  const parts = [];
  parts.push("Averages "+L(Math.min(5,Math.max(0,Math.round(e.mean))))+" tier ("+e.score.toFixed(1)+" out of 5) from "+e.n+(e.n===1?" ballot.":" ballots."));
  if (e.n>=2) {
    parts.push(e.topPct+"% of voters put them in "+L(0)+" or "+L(1)+".");
    parts.push(e.sd<0.7 ? "Voters mostly agree on this one." : e.sd>1.3 ? "Voters are split: some rank them near the top, others far lower." : "There's a bit of disagreement, but the lean is clear.");
  }
  if (S.avg[e.n_]!==undefined) parts.push("About "+S.avg[e.n_]+" average viewers.");
  if (mineRank>0 && mineRank<=MIN_RANK) parts.push("You had them #"+mineRank+".");
  else if (mineRank===0) parts.push("You didn't rank them.");
  return parts.join(" ");
}
function item(num, name, tierIdx, text){
  const c = TIERS[tierIdx][1];
  const d = document.createElement("div"); d.className="item"; d.style.setProperty("--c",c);
  const n = document.createElement("div"); n.className="num"; n.textContent=num;
  const p = document.createElement("div"); p.className="pfp"; fillPfp(p,name);
  const box = document.createElement("div");
  const who = document.createElement("div"); who.className="who";
  const nm = document.createElement("span"); nm.textContent=name;
  const chip = document.createElement("span"); chip.className="chip"; chip.textContent=L(tierIdx);
  who.append(nm,chip);
  const para = document.createElement("p"); para.textContent=text;
  box.append(who,para); d.append(n,p,box); return d;
}
function showResults(){
  show("results");
  const A = agg(), mine = ranking().slice(0,MIN_RANK);
  const nb = allBallots().length;
  $("resInfo").textContent = "Based on "+nb+" submitted "+(nb===1?"ballot":"ballots")+(hasBackend() ? " from everyone who has voted." : " saved on this device. Import other people's ballot codes to widen it.");
  const ml = $("mineList"); ml.textContent="";
  for (let i=mine.length-1;i>=0;i--) { const n=mine[i]; ml.append(item(i+1, n, S.placement[n], pickText(n,i+1,S.placement[n],A))); resolvePfp(n); }
  const entries = Object.entries(A).map(([n,l]) => Object.assign({n_:n}, stats(l)))
    .filter(e => roster().some(r=>r===e.n_))
    .sort((a,b) => b.score-a.score || b.n-a.n || a.sd-b.sd).slice(0,25);
  const cl = $("commList"); cl.textContent="";
  $("commNote").textContent = entries.length ? "Counted down from everyone's ballots. Each write-up comes from the votes: the average tier, how many voters, and how much they agree." : "";
  if (!entries.length) { const d=document.createElement("div"); d.className="empty"; d.textContent="No ballots yet."; cl.append(d); }
  const myRank = n => { const i = mine.indexOf(n); return i>=0 ? i+1 : 0; };
  for (let i=entries.length-1;i>=0;i--) {
    const e = entries[i], ti = Math.min(5,Math.max(0,Math.round(e.mean)));
    cl.append(item(i+1, e.n_, ti, communityText(e, myRank(e.n_)))); resolvePfp(e.n_);
  }
}
